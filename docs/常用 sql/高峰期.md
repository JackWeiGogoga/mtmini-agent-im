# 每个小时订单统计，注意修改下面高峰期时段

```sql
WITH profile AS (
    SELECT *
    FROM mt_shop_peak_window_profile
    WHERE shop_id = 249
      AND day_type = 'WEEKDAY'
),
params AS (
    SELECT
        shop_id,
        lookback_days,
        DATE(generated_at) + INTERVAL 1 DAY AS end_date_exclusive,
        DATE(generated_at) + INTERVAL 1 DAY - INTERVAL lookback_days DAY AS start_date_inclusive
    FROM profile
),
weekday_daily_hour AS (
    SELECT
        DATE(FROM_UNIXTIME(o.order_send_time)) AS order_date,
        HOUR(FROM_UNIXTIME(o.order_send_time)) AS hour_bucket,
        COUNT(1) AS order_count,
        COALESCE(SUM(o.income), 0) AS income_amount
    FROM mt_order o
    JOIN params p ON p.shop_id = o.shop_id
    WHERE o.shop_id = 249
      AND o.status <> 9
      AND o.order_send_time IS NOT NULL
      AND o.order_send_time >= UNIX_TIMESTAMP(p.start_date_inclusive)
      AND o.order_send_time < UNIX_TIMESTAMP(p.end_date_exclusive)
      AND WEEKDAY(FROM_UNIXTIME(o.order_send_time)) BETWEEN 0 AND 4
    GROUP BY
        DATE(FROM_UNIXTIME(o.order_send_time)),
        HOUR(FROM_UNIXTIME(o.order_send_time))
)
SELECT
    order_date,
    CASE WEEKDAY(order_date)
        WHEN 0 THEN '周一'
        WHEN 1 THEN '周二'
        WHEN 2 THEN '周三'
        WHEN 3 THEN '周四'
        WHEN 4 THEN '周五'
    END AS weekday_text,
    CONCAT(LPAD(hour_bucket, 2, '0'), ':00-', LPAD(hour_bucket + 1, 2, '0'), ':00') AS hour_range,
    order_count,
    income_amount,
    CASE
        WHEN hour_bucket BETWEEN 11 AND 13 THEN '命中高峰 11:00-14:00'
        WHEN hour_bucket BETWEEN 19 AND 22 THEN '命中高峰 19:00-23:00'
        ELSE '非高峰'
    END AS peak_evidence
FROM weekday_daily_hour
ORDER BY order_date, hour_bucket;
```

# 计算高峰期，第一部分可以修改营业时间

```sql
WITH RECURSIVE hours AS (
    SELECT 0 AS hour_bucket
    UNION ALL
    SELECT hour_bucket + 1 FROM hours WHERE hour_bucket < 23
),
profile AS (
    SELECT *
    FROM mt_shop_peak_window_profile
    WHERE shop_id = 249
      AND day_type = 'WEEKDAY'
),
params AS (
    SELECT
        shop_id,
        generated_at,
        DATE(generated_at) + INTERVAL 1 DAY AS end_date_exclusive,
        DATE(generated_at) + INTERVAL 1 DAY - INTERVAL lookback_days DAY AS start_date_inclusive
    FROM profile
),
base AS (
    SELECT
        HOUR(FROM_UNIXTIME(o.order_send_time)) AS hour_bucket
    FROM mt_order o
    JOIN params p ON p.shop_id = o.shop_id
    WHERE o.shop_id = 249
      AND o.status <> 9
      AND o.order_send_time IS NOT NULL
      AND o.order_send_time >= UNIX_TIMESTAMP(p.start_date_inclusive)
      AND o.order_send_time < UNIX_TIMESTAMP(p.end_date_exclusive)
      AND o.create_time <= p.generated_at
      AND WEEKDAY(FROM_UNIXTIME(o.order_send_time)) BETWEEN 0 AND 4
),
merged_hour AS (
    SELECT
        hour_bucket,
        COUNT(1) AS order_count
    FROM base
    GROUP BY hour_bucket
),
stats AS (
    SELECT
        GREATEST(ROUND(AVG(order_count), 2), 2) AS candidate_threshold
    FROM merged_hour
    WHERE order_count > 0
)
SELECT
    CONCAT(LPAD(h.hour_bucket, 2, '0'), ':00-', LPAD(h.hour_bucket + 1, 2, '0'), ':00') AS hour_range,
    COALESCE(m.order_count, 0) AS order_count,
    s.candidate_threshold,
    CASE
        WHEN COALESCE(m.order_count, 0) >= s.candidate_threshold THEN '达到高峰阈值'
        ELSE '未达到高峰阈值'
    END AS conclusion
FROM hours h
LEFT JOIN merged_hour m ON m.hour_bucket = h.hour_bucket
CROSS JOIN stats s
ORDER BY h.hour_bucket;
```
