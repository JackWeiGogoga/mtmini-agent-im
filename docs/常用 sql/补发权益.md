```
UPDATE mt_async_task t
JOIN mt_shop s ON s.id = CAST(t.scope_key AS UNSIGNED)
SET
  t.status = 'PENDING',
  t.retry_count = 0,
  t.available_at = NOW(),
  t.claimed_at = NULL,
  t.finished_at = NULL,
  t.worker_id = NULL,
  t.last_error = 'MANUAL_REQUEUE_V6_MAPPING_MISSING',
  t.update_time = NOW()
WHERE t.task_type = 'AUTH_SYNC_RIGHTS'
  AND t.status = 'SUCCESS'
  AND t.create_time >= '2026-05-06 20:30:10'
  AND NOT EXISTS (
    SELECT 1
    FROM mt_shop_right r
    WHERE r.shop_id = s.id
      AND r.service_code = 'H5_PROMOTION'
  );
```

## 审计美团权益是否可能被重复延长或覆盖

以下 SQL 只用于排查，不要直接按查询结果批量修正。历史数据如果已经被旧逻辑重复延长，需要结合订单、授权时间和运营确认单独处理。

按门店查看美团应用订单累计购买天数和当前权益到期时间：

```
SELECT
  s.id AS shop_id,
  s.epoi_id,
  r.service_code,
  r.expire_at,
  r.source_version,
  r.last_order_id,
  COUNT(o.id) AS processed_order_count,
  SUM(o.days) AS processed_days,
  MIN(FROM_UNIXTIME(o.order_time)) AS first_order_time,
  MAX(FROM_UNIXTIME(o.order_time)) AS last_order_time
FROM mt_shop s
JOIN mt_shop_right r
  ON r.shop_id = s.id
LEFT JOIN mt_app_order o
  ON o.shop_id = s.id
 AND o.type = 1
 AND o.version = r.source_version
WHERE r.service_code IN ('H5_PROMOTION', 'VIP_CODE')
GROUP BY
  s.id,
  s.epoi_id,
  r.service_code,
  r.expire_at,
  r.source_version,
  r.last_order_id
ORDER BY r.expire_at DESC;
```

查找仍未应用的美团订单，重点关注缺少必要字段或缺少版本映射的记录：

```
SELECT
  o.id,
  o.order_id,
  o.shop_id,
  o.service_code,
  o.version,
  o.days,
  FROM_UNIXTIME(o.end_time) AS end_time_at,
  o.type,
  CASE
    WHEN o.shop_id IS NULL THEN 'SHOP_ID_MISSING'
    WHEN o.version IS NULL OR o.version = '' THEN 'VERSION_MISSING'
    WHEN o.days IS NULL OR o.days <= 0 THEN 'DAYS_MISSING'
    WHEN o.end_time IS NULL THEN 'END_TIME_MISSING'
    WHEN NOT EXISTS (
      SELECT 1
      FROM mt_version_service_mapping m
      WHERE m.mt_version = o.version
        AND m.enabled = 1
    ) THEN 'VERSION_MAPPING_MISSING'
    ELSE 'READY_TO_APPLY'
  END AS pending_reason
FROM mt_app_order o
WHERE o.type = 0
ORDER BY o.order_time ASC, o.id ASC;
```
